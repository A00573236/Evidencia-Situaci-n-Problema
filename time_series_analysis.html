import pandas as pd

import matplotlib.pyplot as plt

# Upload Excel files
print("Please upload AAPL Excel file")
aapl_upload = files.upload()
aapl_filename = list(aapl_upload.keys())[0]
aapl_df = pd.read_excel(aapl_filename)

print("Please upload JNJ Excel file")
jnj_upload = files.upload()
jnj_filename = list(jnj_upload.keys())[0]
jnj_df = pd.read_excel(jnj_filename)

print("Please upload PEP Excel file")
pep_upload = files.upload()
pep_filename = list(pep_upload.keys())[0]
pep_df = pd.read_excel(pep_filename)

print("\nAAPL DataFrame columns:", list(aapl_df.columns))
print("JNJ DataFrame columns:", list(jnj_df.columns))
print("PEP DataFrame columns:", list(pep_df.columns))

close_column = input("Please enter the column name containing closing prices: ")

try:
    aapl_close = aapl_df[close_column]
    jnj_close = jnj_df[close_column]
    pep_close = pep_df[close_column]
except KeyError:
    print(f"Error: Column '{close_column}' not found in one or more DataFrames")
    print("Please check the column names and try again")
    raise

min_length = min(len(aapl_close), len(jnj_close), len(pep_close))
aapl_close = aapl_close[:min_length]
jnj_close = jnj_close[:min_length]
pep_close = pep_close[:min_length]

series_dict = {
    'AAPL': aapl_close,
    'JNJ': jnj_close,
    'PEP': pep_close
}

diff_dict = {
    name: series.diff().dropna() for name, series in series_dict.items()
}

returns_dict = {
    name: 100 * np.log(series / series.shift(1)).dropna() for name, series in series_dict.items()
}

def unit_root_tests(series, name):
    print(f"\nUnit Root Tests for {name}:")

    adf_result = adfuller(series)
    print("ADF Test:")
    print(f'ADF Statistic: {adf_result[0]:.4f}')
    print(f'p-value: {adf_result[1]:.4f}')
    print(f'Critical Values: {adf_result[4]}')
    print("Interpretation:")
    if adf_result[1] < 0.05:
        print(f"  - p-value < 0.05: Reject null hypothesis - {name} is stationary")
    else:
        print(f"  - p-value >= 0.05: Fail to reject null - {name} may be non-stationary")

    kpss_result = kpss(series)
    print("\nKPSS Test:")
    print(f'KPSS Statistic: {kpss_result[0]:.4f}')
    print(f'p-value: {kpss_result[1]:.4f}')
    print(f'Critical Values: {kpss_result[3]}')
    print("Interpretation:")
    if kpss_result[1] < 0.05:
        print(f"  - p-value < 0.05: Reject null hypothesis - {name} is non-stationary")
    else:
        print(f"  - p-value >= 0.05: Fail to reject null - {name} may be stationary")

for name, series in series_dict.items():
    unit_root_tests(series, name)

plt.figure(figsize=(12,6))
for name, series in series_dict.items():
    plt.plot(series, label=f'{name} Closing Price')
plt.title('Closing Prices for AAPL, JNJ, and PEP')
plt.xlabel('Time')

plt.legend()
plt.grid(True)
plt.show()

plt.legend()
plt.show()

def cointegration_test(df, pair_name):
    result = coint_johansen(df, det_order=0, k_ar_diff=1)
    print(f"\nJohansen Cointegration Test for {pair_name}:")
    print(f"Trace statistic: {result.lr1}")
    print(f"Critical values (90%, 95%, 99%): {result.cvt}")
    print("Interpretation:")
    for i in range(len(result.lr1)):
        if result.lr1[i] > result.cvt[i, 1]:
            print(f"  - r = {i}: Cointegration exists at 95% confidence level")
            print(f"    Trace statistic ({result.lr1[i]:.2f}) > 95% critical value ({result.cvt[i, 1]:.2f})")
        else:
            print(f"  - r = {i}: No cointegration at 95% confidence level")
            print(f"    Trace statistic ({result.lr1[i]:.2f}) <= 95% critical value ({result.cvt[i, 1]:.2f})")
    if result.lr1[0] > result.cvt[0, 1]:
        print(f"Conclusion: {pair_name} are cointegrated - they share a long-run equilibrium relationship")
    else:
        print(f"Conclusion: No evidence of cointegration between {pair_name}")

from itertools import combinations
pairs = list(combinations(series_dict.keys(), 2))
for pair in pairs:
    pair_df = pd.DataFrame({
        pair[0]: series_dict[pair[0]],
        pair[1]: series_dict[pair[1]]
    }).dropna()
    cointegration_test(pair_df, f"{pair[0]} and {pair[1]}")
